cmake_minimum_required(VERSION 3.16)
project(PQC_BIT C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
option(BIT_SOFTWARE_OPS "Build the PC software OP backend" ON)
option(BIT_SANITIZE "Enable ASan and UBSan for tests" OFF)
set(BIT_OPERATOR_SOURCES "" CACHE STRING "Replacement OP backend sources")
set(BIT_OPERATOR_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "OP interface include directory")
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(CORE "${ROOT}/core/common")
set(MODULES randombytes ntt packing parabola poly polyvec_ref sample sign spectral symmetric signature_api signature_codec_128)
set(CORE_SOURCES)
foreach(name IN LISTS MODULES)
    list(APPEND CORE_SOURCES "${CORE}/${name}.c")
endforeach()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Wpedantic)
    if(BIT_SANITIZE)
        add_compile_options(-O1 -g -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
function(bit_includes target)
    target_include_directories(${target} PRIVATE "${ROOT}/include" "${CORE}" "${BIT_OPERATOR_INCLUDE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/soft" "${CMAKE_CURRENT_SOURCE_DIR}/test" "${CMAKE_CURRENT_SOURCE_DIR}/test/reference")
endfunction()
add_library(pqc_bit STATIC ${CORE_SOURCES})
bit_includes(pqc_bit)
target_include_directories(pqc_bit PUBLIC "${ROOT}/include")
if(BIT_SOFTWARE_OPS)
    target_sources(pqc_bit PRIVATE soft/op_hash.c soft/op_trng.c soft/op_ntt.c soft/sha256.c soft/sm3.c soft/endian.c)
elseif(BIT_OPERATOR_SOURCES)
    target_sources(pqc_bit PRIVATE ${BIT_OPERATOR_SOURCES})
endif()
if(WIN32)
    target_link_libraries(pqc_bit PUBLIC bcrypt)
else()
    target_link_libraries(pqc_bit PUBLIC m)
endif()
if(BUILD_TESTING)
    add_library(bit_kat_drng OBJECT test/reference/drng.c)
    bit_includes(bit_kat_drng)
    add_library(bit_kat_random OBJECT test/reference/op_trng_kat.c)
    bit_includes(bit_kat_random)
    add_executable(test_bit_correctness test/test_correctness.c test/reference/fips202.c $<TARGET_OBJECTS:bit_kat_drng> $<TARGET_OBJECTS:bit_kat_random>)
    bit_includes(test_bit_correctness)
    target_link_libraries(test_bit_correctness PRIVATE pqc_bit)
    add_test(NAME bit_correctness COMMAND test_bit_correctness)
    add_executable(test_bit_production test/test_correctness.c test/reference/fips202.c $<TARGET_OBJECTS:bit_kat_drng>)
    bit_includes(test_bit_production)
    target_link_libraries(test_bit_production PRIVATE pqc_bit)
    add_test(NAME bit_production_correctness COMMAND test_bit_production)

    add_library(bit_memory_core OBJECT ${CORE_SOURCES})
    bit_includes(bit_memory_core)
    target_compile_definitions(bit_memory_core PRIVATE malloc=bit_test_malloc free=bit_test_free)
    target_compile_options(bit_memory_core PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/test/memory_probe.h")
    add_executable(test_bit_memory test/test_memory.c test/memory_probe.c $<TARGET_OBJECTS:bit_memory_core>)
    bit_includes(test_bit_memory)
    target_link_libraries(test_bit_memory PRIVATE pqc_bit)
    add_test(NAME bit_memory COMMAND test_bit_memory)

    if(BIT_SOFTWARE_OPS)
        add_library(bit_fault_hash OBJECT soft/op_hash.c)
        bit_includes(bit_fault_hash)
        target_compile_definitions(bit_fault_hash PRIVATE OP_hash=real_OP_hash OP_hash_init=real_OP_hash_init OP_hash_absorb=real_OP_hash_absorb OP_hash_squeeze=real_OP_hash_squeeze)
        add_executable(test_bit_op test/test_op_contract.c test/memory_probe.c test/reference/fips202.c $<TARGET_OBJECTS:bit_memory_core> $<TARGET_OBJECTS:bit_fault_hash> $<TARGET_OBJECTS:bit_kat_drng>)
        bit_includes(test_bit_op)
        target_link_libraries(test_bit_op PRIVATE pqc_bit)
        add_test(NAME bit_op_contract COMMAND test_bit_op)
        add_executable(test_bit_random test/test_random_operator.c test/memory_probe.c $<TARGET_OBJECTS:bit_memory_core>)
        bit_includes(test_bit_random)
        target_link_libraries(test_bit_random PRIVATE pqc_bit)
        add_test(NAME bit_random_contract COMMAND test_bit_random)
        add_library(bit_ntt_real OBJECT soft/op_ntt.c)
        bit_includes(bit_ntt_real)
        target_compile_definitions(bit_ntt_real PRIVATE OP_ntt256_init=real_OP_ntt256_init OP_ntt256=real_OP_ntt256 OP_intt256=real_OP_intt256 OP_cwm=real_OP_cwm)
        add_library(bit_ntt_generic OBJECT test/reference/op_ntt_generic.c)
        bit_includes(bit_ntt_generic)
        foreach(backend IN ITEMS real generic)
            add_executable(test_bit_ntt_${backend} test/test_ntt_contract.c test/memory_probe.c test/reference/ntt_scalar.c $<TARGET_OBJECTS:bit_memory_core> $<TARGET_OBJECTS:bit_kat_drng> $<TARGET_OBJECTS:bit_kat_random> $<TARGET_OBJECTS:bit_ntt_${backend}>)
            bit_includes(test_bit_ntt_${backend})
            target_link_libraries(test_bit_ntt_${backend} PRIVATE pqc_bit)
            add_test(NAME bit_ntt_${backend} COMMAND test_bit_ntt_${backend})
        endforeach()
    endif()
endif()
add_executable(bit_kat EXCLUDE_FROM_ALL test/KAT_SIG.c test/reference/drng.c test/reference/op_trng_kat.c)
bit_includes(bit_kat)
target_link_libraries(bit_kat PRIVATE pqc_bit)
